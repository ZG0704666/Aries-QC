function ajaxGet(url, setting) {
  jQuery.support.cors = true;
  var opt = {
    async: true,
    type: 'text/html',
    contentType: 'application/x-www-form-urlencoded',

    success: function () {},
    complete: function () {},
    data: null,
    xhrFields: {
      withCredentials:true
    }
  };
  opt = $.extend(opt, setting);
  $.ajax({
    type: "get",
    url: url,
    dataType: opt.type,
    error: opt.error,
    contentType: opt.contentType,
    data: opt.data,
    success: opt.success,
    complete: opt.complete
  });
}
